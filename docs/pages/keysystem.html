Key System & Config

Full WindUI-parity features in v2.0.0

Key System

Lua
local Window = VillainsUI:CreateWindow({
    Title = "My Hub",
    Folder = "MyHub",
    KeySystem = {
        Key = { "KEY-1", "KEY-2" },
        SaveKey = true,
        Note = "Get key from our Discord!",
        URL = "https://your-key-link.com",
        KeyValidator = function(key)
            return key == "custom-valid-key"
        end,
        API = {
            { Type = "luarmor", ScriptId = "id", Discord = "https://discord.gg/..." },
            { Type = "platoboost", ServiceId = 1234, Secret = "secret" },
            { Type = "pandadevelopment", ServiceId = "hub-name" },
        },
    },
})

Config Save / Load

Lua
Tab:Toggle({ Title = "Auto Farm", Flag = "AutoFarm", Default = false })
Tab:Slider({ Title = "Speed", Flag = "Speed", Min = 16, Max = 200, Default = 16 })

Tab:Button({
    Title = "Save",
    Callback = function() Window:SaveConfig("default") end,
})

Tab:Button({
    Title = "Load",
    Callback = function() Window:LoadConfig("default") end,
})

Localization

Lua
VillainsUI:Localization({
    Enabled = true,
    Prefix = "loc:",
    DefaultLanguage = "en",
    Translations = {
        ["en"] = { ["WELCOME"] = "Welcome!" },
        ["id"] = { ["WELCOME"] = "Selamat datang!" },
    },
})

VillainsUI:SetLanguage("id")
Window:Tab({ Title = "loc:WELCOME" })
